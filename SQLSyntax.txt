-- 3. Create the smart Trigger Function
CREATE OR REPLACE FUNCTION log_vehicle_booking_to_trip_logs()
RETURNS trigger AS $$
DECLARE
  v_driver_name text;
  v_origin text;
  v_destination text;
  v_status text;
BEGIN
  -- Only log when status changes
  IF OLD.status IS DISTINCT FROM NEW.status THEN

    -- 🎯 SMART LOOKUP 1: Check the actual booking first!
    -- This ignores vehicle unassignments and looks directly at who was booked.
    IF NEW.driver_id IS NOT NULL THEN
        SELECT name INTO v_driver_name 
        FROM public.driver_profiles 
        WHERE id = NEW.driver_id;
    END IF;

    -- 🎯 SMART LOOKUP 2: Active Vehicle Assignment fallback
    IF v_driver_name IS NULL THEN
        SELECT name INTO v_driver_name 
        FROM public.driver_profiles 
        WHERE assigned_vehicle_id = NEW.vehicle_id 
        LIMIT 1;
    END IF;

    -- 🎯 SMART LOOKUP 3: Active Fleet Assignment fallback
    IF v_driver_name IS NULL THEN
        SELECT dp.name INTO v_driver_name
        FROM public.driver_profiles dp
        JOIN public.vehicles v ON v.fleet_id = dp.assigned_fleet_id
        WHERE v.id = NEW.vehicle_id
        LIMIT 1;
    END IF;

    -- 🎯 SMART LOOKUP 4: HISTORICAL RECOVERY (The Magic Fix for "Returned" trips)
    -- If React already cleared the driver assignment, grab the last known name from the previous log!
    IF v_driver_name IS NULL THEN
        SELECT driver_name INTO v_driver_name
        FROM public.trip_logs
        WHERE vehicle_id = NEW.vehicle_id 
          AND driver_name IS NOT NULL 
          AND driver_name != 'Unknown Driver'
        ORDER BY created_at DESC
        LIMIT 1;
    END IF;

    -- Final Fallback
    IF v_driver_name IS NULL THEN
        v_driver_name := 'Unknown Driver';
    END IF;

    -- Determine the correct Status, Origin, and Destination (The DAR X Logic)
    IF NEW.status = 'Approved' THEN
        v_status := 'Scheduled';
        v_origin := 'DAR X';
        v_destination := NEW.destination;
    ELSIF NEW.status = 'Ongoing' THEN
        v_status := 'Ongoing';
        v_origin := 'DAR X';
        v_destination := NEW.destination;
    ELSIF NEW.status = 'Returned' THEN
        v_status := 'Completed';
        -- 🔄 SWAP: When returning, they leave the Destination and head back to DAR X
        v_origin := NEW.destination; 
        v_destination := 'DAR X';
    ELSIF NEW.status = 'Rejected' THEN
        v_status := 'Cancelled';
        v_origin := 'DAR X';
        v_destination := NEW.destination;
    ELSE
        v_status := 'Scheduled';
        v_origin := 'DAR X';
        v_destination := NEW.destination;
    END IF;

    -- Insert into trip_logs
    INSERT INTO public.trip_logs (
      vehicle_id,
      driver_name,    -- Perfectly preserved name
      origin,
      destination,
      purpose,
      status,
      start_datetime,
      end_datetime,
      created_at
    )
    VALUES (
      NEW.vehicle_id,
      v_driver_name,  
      v_origin,
      v_destination,
      NEW.purpose,
      v_status,
      NEW.start_datetime,
      NEW.end_datetime,
      now()
    );

  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 4. Re-attach the trigger
DROP TRIGGER IF EXISTS trg_vehicle_booking_trip_logs ON public.vehicle_bookings;

CREATE TRIGGER trg_vehicle_booking_trip_logs
AFTER UPDATE OF status ON public.vehicle_bookings
FOR EACH ROW
EXECUTE FUNCTION log_vehicle_booking_to_trip_logs();

-- 2. CREATE THE AUTO-REJECT FUNCTION
-- This function finds 'Pending' bookings where the start time has passed and rejects them
CREATE OR REPLACE FUNCTION auto_reject_expired_bookings()
RETURNS void AS $$
BEGIN
  UPDATE public.vehicle_bookings
  SET 
    status = 'Rejected',
    -- Appends a system note so you know why it was rejected, while keeping any existing notes
    admin_notes = CASE 
        WHEN admin_notes IS NULL OR admin_notes = '' 
        THEN '[System]: Auto-rejected because the departure time passed without admin approval.'
        ELSE admin_notes || ' | [System]: Auto-rejected because the departure time passed without admin approval.'
    END
  WHERE 
    status = 'Pending' 
    AND start_datetime <= now();
END;
$$ LANGUAGE plpgsql;

-- 3. SCHEDULE THE BACKGROUND JOB
-- This tells the database to run the function above every 1 minute ('* * * * *')
SELECT cron.schedule(
  'auto-reject-expired-bookings-job', 
  '* * * * *',                        
  'SELECT auto_reject_expired_bookings();' 
);
